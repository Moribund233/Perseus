4import { Repository, Branch, Commit } from './types';
5import { EventBus } from '../events';
6import { createLogger } from './utils';
7
8const logger = createLogger('forge:core');
9
10/**
11 * ForgeEngine is the central orchestrator for repository operations.
12 * It manages branches, commits, merges, and coordinates with
13 * the storage backend and event system.
14 */
15export class ForgeEngine {
16 private repos: Map<string, Repository> = new Map();
17 private bus: EventBus;
18
19 constructor(private opts: EngineOptions) {
20 this.bus = new EventBus(opts.eventConfig);
21 logger.info('ForgeEngine initialized', { version: opts.version });
22 }
23
24 /**
25 * Create a new repository in the forge.
26 */
27 async createRepository(
28 name: string,
29 owner: string,
30 config?: Partial<RepoConfig>
31 ): Promise<Repository> {
32 const existing = await this.findRepo(name, owner);
33 if (existing) {
34 throw new RepoError('REPO_EXISTS', `Repository "${name}" already exists`);
35 }
36
37 const repo = await this.storage.createRepo({
38 name,
39 owner,
40 visibility: config?.visibility ?? 'public',
41 initialized: false,
42 createdAt: new Date(),
43 });
44
45 await this.bus.publish('repo:created', { repo });
46 this.repos.set(repo.id, repo);
47
48 logger.info('Repository created', { name, owner, id: repo.id });
49 return repo;
50 }
51
52 /**
53 * Clone a repository by fetching all objects from remote.
54 */
55 async clone(
56 remoteUrl: string,
57 opts?: CloneOptions
58 ): Promise<Repository> {
59 const parsed = this.parseRemoteUrl(remoteUrl);
60 const refs = await this.transport.fetchRefs(parsed.host, parsed.path);
61
62 if (!refs.length) {
63 throw new RepoError('EMPTY_REPO', 'Remote has no refs');
64 }
65
66 const repo = await this.storage.initRepo({
67 name: parsed.repoName,
68 remoteUrl,
69 bare: opts?.bare ?? true,
70 });
71
72 for (const ref of refs) {
73 await this.storage.writeRef(repo.id, ref);
74 }
75
76 await this.bus.publish('repo:cloned', { repo });
77 return repo;
78 }
79
80 /**
81 * Create a new branch from a source ref.
82 */
83 async createBranch(
84 repoId: string,
85 name: string,
86 sourceRef: string
87 ): Promise<Branch> {
88 const targetCommit = await this.storage.resolveRef(repoId, sourceRef);
89 if (!targetCommit) {
90 throw new RepoError('REF_NOT_FOUND', `Ref "${sourceRef}" not found`);
91 }
92
93 const branch = await this.storage.createBranch({
94 repoId, name,
95 target: targetCommit.oid,
96 });
97
98 return branch;
99 }
100
101 /**
102 * Merge source branch into target branch.
103 */
104 async merge(
105 repoId: string,
106 source: string,
107 target: string,
108 author: Author
109 ): Promise<Commit> {
110 const sourceTip = await this.storage.getRefTip(repoId, source);
111 const targetTip = await this.storage.getRefTip(repoId, target);
112
113 const ancestor = await this.storage.mergeBase(
114 repoId, sourceTip.oid, targetTip.oid
115 );
116
117 const mergeCommit = await this.storage.createMergeCommit({
118 repoId, source: sourceTip, target: targetTip,
119 ancestor: ancestor?.oid,
120 author,
121 message: `Merge branch '${source}' into '${target}'`,
122 });
123
124 await this.bus.publish('ref:updated', {
125 ref: target, oid: mergeCommit.oid
126 });
127
128 return mergeCommit;
129 }
130
131 /**
132 * List all repositories accessible to a user.

134 async listRepositories(

144
145 return repos;
146 }

148 /**
149 * Find a repository by name and owner.
150 */
151 private async findRepo(
152 name: string,
153 owner: string
154 ): Promise<Repository | null> {
155 const repo = await this.storage.findByName(name, owner);
156 return repo ?? null;

159 /**
160 * Parse a remote URL into its components.
161 */
162 private parseRemoteUrl(url: string): RemoteInfo {
163 const pattern = /^(?:https?:\/\/|git@)([^:\/]+)[:\/](.+)\/(.+)\.git$/;
164 const match = url.match(pattern);
165
166 if (!match) {
167 throw new RepoError('INVALID_URL', `Invalid remote URL: "${url}"`);
168 }
169
170 return {
171 host: match[1],
172 path: `${match[2]}/${match[3]}`,
173 repoName: match[3],
174 };
175 }
176
177 /**
178 * Get engine health status.
179 */
180 async health(): Promise<HealthStatus> {
181 const storageOk = await this.storage.ping();
182 const busOk = await this.bus.ping();
183
184 return { status: storageOk && busOk ? 'healthy' : 'degraded' };
185 }
